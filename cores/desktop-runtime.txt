# The freedesktop Flatpak runtime the Linux desktop build targets
# (docs/dev/desktop-cores.md). cores/build-desktop.sh builds linux-x86_64 cores
# inside sdk=; the SteamOS Flatpak must use the matching runtime=.
# freedesktop-sdk release/26.08 (current stable branch, checked 2026-10-17).
sdk=org.freedesktop.Sdk/x86_64/26.08
runtime=org.freedesktop.Platform/x86_64/26.08
# The portable tar.gz is built in an older SDK so it runs on older distros
# (tools/release/build-linux.sh; docs/dev/release-linux.md, "Oldest supported
# system"). 22.08 is end-of-life but still on Flathub and has glibc 2.35 and
# GCC 12; clang for the Flutter runner comes from its LLVM extension. The
# build fails when a portable binary needs more than portable_glibc= or
# portable_glibcxx= (libstdc++ of GCC 12).
portable_sdk=org.freedesktop.Sdk/x86_64/22.08
portable_llvm=org.freedesktop.Sdk.Extension.llvm16/x86_64/22.08
portable_glibc=2.35
portable_glibcxx=3.4.30
